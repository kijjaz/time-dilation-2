\begin{abstract}
This document formulates the 32-band logarithmic frequency estimation, peak ballistics decay dynamics, and 2D parametric state phase space trajectory plotting powering the \texttt{[spectrometer\textasciitilde{}]} and \texttt{[time.xy]} visualizer node objects in \textbf{Time Dilation DAW}.
\end{abstract}

\section{Logarithmic Frequency Band Spacing}
Human auditory perception of pitch follows logarithmic frequency scaling (octaves). In the 32-band live spectrum estimator \texttt{[spectrometer\textasciitilde{}]}, band center frequencies $f_b$ for $b \in [0, 31]$ are spaced logarithmically between $f_{\text{min}} = 20\,\text{Hz}$ and $f_{\text{max}} = 20\,000\,\text{Hz}$:
\begin{equation}
f_b = f_{\text{min}} \cdot \left( \frac{f_{\text{max}}}{f_{\text{min}}} \right)^{\frac{b}{B-1}}, \quad \text{where } B = 32
\end{equation}

\section{Short-Time Energy \& Peak Ballistics Decay}
For each block of audio input $x(n)$ of length $K$:
\begin{equation}
E_{\text{raw}} = \max_{n=0}^{K-1} |x(n)|
\end{equation}
For band $b$, band magnitude level $M_b(k)$ at block $k$ is smoothed using 1-pole lowpass filter:
\begin{equation}
M_b(k) = 0.75 \cdot M_b(k-1) + 0.25 \cdot T_b(k)
\end{equation}
where target $T_b(k) = E_{\text{raw}} \cdot \left(1 - 0.4 \frac{b}{B-1}\right) \cdot \Phi_b(t)$.

\subsection{Peak Indicator Hold \& Exponential Decay}
Peak indicator dot levels $P_b(k)$ catch instant maxima and decay exponentially:
\begin{equation}
P_b(k) =
\begin{cases}
M_b(k), & M_b(k) \ge P_b(k-1) \\
0.94 \cdot P_b(k-1), & M_b(k) < P_b(k-1)
\end{cases}
\end{equation}

\section{2D Phase Space Parametric Trajectories (\texttt{[time.xy]})}
The \texttt{[time.xy]} oscilloscope node visualizes the 2D parametric state space trajectory $\mathbf{r}(t) = (x(t), y(t))^T$ generated by two audio or control signal streams:
\begin{equation}
\mathbf{r}[n] = \begin{bmatrix} x[n] \\ y[n] \end{bmatrix} \in [-2.0, 2.0] \times [-2.0, 2.0]
\end{equation}
Points are stored in a 256-sample ring buffer:
\begin{equation}
\mathcal{P} = \{ \mathbf{r}[n-255], \dots, \mathbf{r}[n-1], \mathbf{r}[n] \}
\end{equation}
Rendering a continuous vector trace path on the dark slate canvas matrix allows real-time visualization of Lissajous curves, phase relationships, and relativistic coordinate time orbits.
