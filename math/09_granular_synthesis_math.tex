\begin{abstract}
This paper formulates the Gabor granular time-stretching algorithms, grain window envelopes, stochastic Poisson grain distributions, and overlap-add (OLA) sum equations powering the \texttt{[sampler\textasciitilde{}]} audio node.
\end{abstract}

\section{Gabor Grain Representation}
A single audio grain $g_k(t)$ of length $L_k$ centered at buffer offset $\tau_k$ with pitch playback speed $S_k$ is defined by modulating an audio sample buffer $x(t)$ with a smooth window envelope $w(t)$:
\begin{equation}
g_k(t) = w\left(\frac{t - t_{k, \text{start}}}{L_k}\right) \cdot x\left(\tau_k + S_k \cdot (t - t_{k, \text{start}})\right)
\end{equation}
where $t_{k, \text{start}}$ is the physical time instant when grain $k$ is spawned.

\section{Grain Windowing Envelope Functions}
To eliminate boundary click artifacts at grain edges ($t = 0$ and $t = L_k$), window function $w(\eta)$ for $\eta \in [0, 1]$ is selected:

\subsection{Hanning Window Envelope}
\begin{equation}
w_{\text{Hanning}}(\eta) = \frac{1}{2} \left( 1 - \cos\left(2\pi \eta\right) \right) = \sin^2(\pi \eta)
\end{equation}

\subsection{Gaussian Window Envelope}
\begin{equation}
w_{\text{Gaussian}}(\eta) = \exp\left( -\frac{(\eta - 0.5)^2}{2\sigma^2} \right), \quad \text{where } \sigma \approx 0.16
\end{equation}

\section{Stochastic Poisson Grain Spawning Distribution}
The inter-grain arrival interval $\Delta t_{\text{spawn}}$ is governed by a Poisson point process with mean grain density rate $\lambda$ (grains per second):
\begin{equation}
P(\Delta t_{\text{spawn}} = \delta) = \lambda e^{-\lambda \delta}
\end{equation}
In discrete time, next grain spawn time $n_{\text{next}}$ is randomized using uniform random variate $U \in (0, 1)$:
\begin{equation}
n_{\text{next}} = n_{\text{current}} + \left\lfloor -\frac{f_s}{\lambda} \ln(U) \right\rfloor
\end{equation}

\section{Overlap-Add (OLA) Grain Summation}
The continuous output signal $y(t)$ is generated by summing all active overlapping grains $g_k(t) \in \mathcal{G}_{\text{active}}$:
\begin{equation}
y(t) = \sum_{k \in \mathcal{G}_{\text{active}}} g_k(t)
\end{equation}
Under uniform grain overlap with Hanning windowing, constant power energy normalization $\sum w_k(t)^2 = 1.0$ is maintained, preserving original dynamic loudness across variable grain densities.
